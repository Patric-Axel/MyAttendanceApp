drop database if exists MyAttendanceApp;
create database MyAttendanceApp;
use MyAttendanceApp;

create table tb_roles(
idrol int not null primary key,
descripcion varchar(15)
);

create table tb_estado_usu(
idestado int not null primary key,
descripcion varchar(50)
);

create table tb_estado_asistencia(
idestadoasis int not null primary key,
descripcion varchar(50)
);

create table tb_usuarios(
idusuario int not null auto_increment primary key,
nombres varchar(50) not null,
apellidos varchar(50) not null,
direccion varchar(100),
email varchar(50) unique not null,
celular varchar(15) not null,
pwd varchar(255) not null,
fotoperfil varchar(100),
idestado int,
idrol int,
fecha date default (curdate()),
constraint fk_estado foreign key(idestado) references tb_estado_usu(idestado),
constraint fk_rol foreign key(idrol) references tb_roles(idrol)
);

create table tb_asistencia(
idasistencia int primary key auto_increment,
fecha date default (curdate()),
fecharegistro datetime default current_timestamp,
horaentrada time,
horasalida time,
horastrabajadas decimal(5,2),
latitudentrada decimal (10,8),
longitudentrada decimal (11,8),
latitudsalida decimal (10,8),
longitudsalida decimal (11,8),
direntrada varchar(200),
dirsalida varchar(200),
fotoentrada varchar(200),
fotosalida varchar(200),
idestadoasis int,
idusuario int,
constraint fk_estadoasistencia foreign key(idestadoasis) references tb_estado_asistencia(idestadoasis),
constraint fk_usuarios foreign key(idusuario) references tb_usuarios(idusuario),
constraint uk_asistencia unique(idusuario, fecha)
);
